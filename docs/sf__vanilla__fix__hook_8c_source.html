775{

777 uint16_t owner = _this->CGdFigure->figures[figure_id].owner;
778 if ((owner != 0) && (race != 0) && (race < 7))

781 {
782 return 0;

785 {
786 if (!figureAPI.isWarrior(_this->CGdFigure, figure_id))
787 {
788 uint16_t worker_count = _this->CGdPlayer->players[owner].civilian_count[race - 1];
789 if (worker_count > 0)
790 {
791 worker_count--;
792 }
793 _this->CGdPlayer->players[owner].civilian_count[race - 1] = worker_count;
794 return 0;
795 }
796 else
797 {
798 bool side = ((race != 3) && (race != 2) && (race != 1));
799 decreaseUsedArmyUnitCount(_this->CGdPlayer, owner, side);
800 if (figureAPI.isSiegeUnit(_this->CGdFigure, figure_id))
801 {
802 decreaseUsedArmyUnitCount(_this->CGdPlayer, owner, side);
803 decreaseUsedArmyUnitCount(_this->CGdPlayer, owner, side);

805 }
806 }
807
808 }
