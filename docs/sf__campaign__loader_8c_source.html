18static bool json_token_streq(const char *json, const jsmntok_t *token, const char *s)
19{
20 return token->type == JSMN_STRING &&
21 (int)strlen(s) == token->end - token->start &&
22 strncmp(json + token->start, s, token->end - token->start) == 0;
23}

49static int skip_token_tree(const jsmntok_t *tokens, int token_count, int index)
50{
51 if (index >= token_count)
52 {
53 return index;
54 }
55
56 if (tokens[index].type == JSMN_OBJECT || tokens[index].type == JSMN_ARRAY)
57 {
58 int children_count = tokens[index].size;
59 int next_index = index + 1;
60 for (int i = 0; i < children_count; i++)
61 {
62 if (next_index >= token_count)
63 {
64 return token_count;
65 }
66 if (tokens[index].type == JSMN_OBJECT)
67 {
68 next_index = skip_token_tree(tokens, token_count, next_index);
69 next_index = skip_token_tree(tokens, token_count, next_index);
70 }
71 else
72 {
73 next_index = skip_token_tree(tokens, token_count, next_index);
74 }
75 }
76 return next_index;
77 }
78
79 return index + 1;
80}

82static CampaignFieldKey parse_campaign_field_key(const char *json, const jsmntok_t *token)
83{

111static bool parse_campaign_from_tokens(const char *json, const jsmntok_t *tokens,
112 int token_count, CampaignJson *out_campaign)
113{
114 if (token_count < 1 || tokens[0].type != JSMN_OBJECT)
115 {
116 return false;
117 }
118
119 memset(out_campaign, 0, sizeof(CampaignJson));
120
121 /* -1 means "unset" to every consumer of avatar_type. */
122 out_campaign->def.avatar_type = -1;
123
124 int pairs_to_process = tokens[0].size;
125 int current_token_index = 1;
126
127 for (int i = 0; i < pairs_to_process; i++)
128 {
129 if (current_token_index + 1 >= token_count)
130 {
131 return false;
132 }
133
134 CampaignFieldKey key = parse_campaign_field_key(json, &tokens[current_token_index]);
135 const jsmntok_t *value_token = &tokens[current_token_index + 1];
136 SFSF_CampaignDef *def = &out_campaign->def;
137
138 switch (key)
139 {
140 case CAMPAIGN_FIELD_NAME:
141 json_str(json, value_token, def->name, sizeof(def->name));
142 out_campaign->found_name = true;
143 break;
144 case CAMPAIGN_FIELD_CAMPAIGN_NAME_ID:
145 {
146 int campaign_name_id = jsonint(json, value_token);
147 def->campaign_name_id = campaign_name_id;
148 out_campaign->found_campaign_name_id = true;
149 break;
150 }
151 case CAMPAIGN_FIELD_DESCRIPTION:
152 json_str(json, value_token, def->description, sizeof(def->description));
153 out_campaign->found_description = true;
154 break;
155 case CAMPAIGN_FIELD_AUTHOR:
156 json_str(json, value_token, def->author, sizeof(def->author));
157 out_campaign->found_author = true;
158 break;
159 case CAMPAIGN_FIELD_START_MAP:
160 json_str(json, value_token, def->start_map, sizeof(def->start_map));
161 out_campaign->found_start_map = true;
162 break;
163 case CAMPAIGN_FIELD_TUTORIAL_MAP:
164 json_str(json, value_token, def->tutorial_map, sizeof(def->tutorial_map));
165 out_campaign->found_tutorial_map = true;
166 break;
167 case CAMPAIGN_FIELD_CAMPAIGN_FOLDER:
168 json_str(json, value_token, def->campaign_folder, sizeof(def->campaign_folder));
169 out_campaign->found_campaign_folder = true;
170 break;
171 case CAMPAIGN_FIELD_STARTERKIT:
172 json_str(json, value_token, def->starterkit, sizeof(def->starterkit));
173 out_campaign->found_starterkit = true;
174 break;
175 case CAMPAIGN_FIELD_INTRO_VIDEO:
176 json_str(json, value_token, def->intro_video, sizeof(def->intro_video));
177 out_campaign->found_intro_video = true;
178 break;
179 case CAMPAIGN_FIELD_AVATAR_TYPE:
180 {
181 int avatar_type = jsonint(json, value_token);
182 def->avatar_type = avatar_type;
183 out_campaign->found_avatar_type = true;
184 break;
185 }
186 default:
187 // Unknown key - skipped so future fields stay backwards compatible.
188 break;
189 }
190
191 current_token_index = skip_token_tree(tokens, token_count, current_token_index); // key
192 current_token_index = skip_token_tree(tokens, token_count, current_token_index); // value
193 }
194
195 return true;
196}

198bool parse_campaign_json_file(const char *path, CampaignJson *out_campaign)
199{
200 char *json_string = readfile(path);
201 if (json_string == NULL)
202 {
203 // readfile has already logged the reason.
204 return false;
205 }
206
207 jsmn_parser parser;
208 jsmn_init(&parser);
209
210 size_t json_length = strlen(json_string);
211 int token_count = jsmn_parse(&parser, json_string, json_length, NULL, 0);
212 if (token_count < 1)
213 {
214 log_error("Campaign JSON is not valid JSON: %s", path);
215 log_debug(DEBUG_HIGH, json_string);
216 free(json_string);
217 return false;
218 }
219
220 jsmntok_t *tokens = (jsmntok_t *)calloc(token_count, sizeof(jsmntok_t));
221 if (tokens == NULL)
222 {
223 log_error("Unable to allocate %d JSON tokens for: %s", token_count, path);
224 free(json_string);
225 return false;
226 }
227
228 jsmn_init(&parser);
229 token_count = jsmn_parse(&parser, json_string, json_length, tokens, token_count);
230
231 bool success = false;
232 if (token_count < 1)
233 {
234 log_error("Campaign JSON failed to tokenize: %s", path);
235 log_debug(DEBUG_HIGH, json_string);
236 }
237 else if (tokens[0].type != JSMN_OBJECT)
238 {
239 log_error("Campaign JSON must be a single JSON object: %s", path);
240 }
241 else
242 {
243 success = parse_campaign_from_tokens(json_string, tokens, token_count, out_campaign);
244 if (!success)
245 {
246 log_error("Campaign JSON structure invalid: %s", path);
247 }
248 }
249
250 free(tokens);
251 free(json_string);
252 return success;
253}
